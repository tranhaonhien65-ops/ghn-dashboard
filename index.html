Cài đặt tiện ích Tampermonkey trên trình duyệt Cốc Cốc / Chrome → Dán đoạn mã bên dưới vào. Mỗi khi bạn mở tab baocao.ghn.vn, số liệu sẽ tự động đẩy về Dashboard ngầm mà bạn không cần bấm bất kỳ nút nào! 

// ==UserScript==
// @name         GHN Dashboard Auto-Sync Master 5.0
// @namespace    http://tampermonkey.net/
// @version      5.0
// @description  Tự động bắt gói tin và đồng bộ 100% số liệu Metabase từ baocao.ghn.vn về GHN Dashboard
// @author       Hao Nhien
// @match        https://baocao.ghn.vn/*
// @match        https://data-bi.ghn.vn/*
// @allFrames    true
// @run-at       document-start
// @grant        GM_xmlhttpRequest
// @connect      baocao-v2-api.ghn.vn
// @connect      data-bi.ghn.vn
// @connect      ghn-dashboard.onrender.com
// @connect      localhost
// @connect      127.0.0.1
// ==/UserScript==

(function() {
    'use strict';

    function showFloatingBadge(msg, color = '#059669') {
        let b = document.getElementById('ghn-master-badge');
        if (!b) {
            b = document.createElement('div');
            b.id = 'ghn-master-badge';
            b.style.cssText = 'position:fixed;bottom:20px;right:20px;z-index:9999999;padding:12px 18px;border-radius:12px;color:#fff;font-family:sans-serif;font-size:13px;font-weight:bold;box-shadow:0 8px 25px rgba(0,0,0,0.3);display:flex;align-items:center;gap:8px;transition:all 0.3s;';
            document.body.appendChild(b);
        }
        b.style.background = color;
        b.innerHTML = msg;
    }

    function sendToDashboard(payload) {
        const endpoints = [
            'https://ghn-dashboard.onrender.com/api/token/update',
            'http://localhost:8080/api/token/update'
        ];
        endpoints.forEach(url => {
            GM_xmlhttpRequest({
                method: 'POST',
                url: url,
                headers: { 'Content-Type': 'application/json' },
                data: JSON.stringify(payload),
                onload: function(r) {
                    try {
                        const res = JSON.parse(r.responseText);
                        const count = res.total_orders || payload.rows?.length || 0;
                        if (count > 0) {
                            showFloatingBadge(`🎉 Đã đồng bộ ${count.toLocaleString('vi-VN')} đơn về Dashboard!`, '#059669');
                        } else {
                            showFloatingBadge(`🔑 Đã cập nhật Token! Đang tải danh sách đơn...`, '#0068a7');
                        }
                    } catch(e) {
                        showFloatingBadge(`🎉 Đã đẩy dữ liệu thành công!`, '#059669');
                    }
                },
                onerror: function(e) {
                    console.warn('[GHN Sync] Gửi thất bại tới ' + url, e);
                }
            });
        });
    }

    function fetchDirectCard4885(jwt, tk) {
        if (!jwt || window.__is_fetching_card_4885) return;
        window.__is_fetching_card_4885 = true;
        showFloatingBadge(`⏳ Đang tải toàn bộ dữ liệu đơn từ Metabase...`, '#0068a7');
        
        const cardUrl = `https://data-bi.ghn.vn/api/embed/dashboard/${jwt}/dashcard/6243/card/4885?parameters=%7B%22kho%22%3Anull%2C%22ma_kho%22%3A%5B%2220335000%22%5D%2C%22phan_loai%22%3Anull%7D`;
        GM_xmlhttpRequest({
            method: 'GET',
            url: cardUrl,
            headers: {
                'Accept': 'application/json',
                'Content-Type': 'application/json',
                'x-metabase-client': 'embedding-simple',
                'x-metabase-embed-referrer': 'https://baocao.ghn.vn/dashboards/6a71852ca20df68be00411c2',
                'x-metabase-locale': 'en'
            },
            onload: function(r) {
                window.__is_fetching_card_4885 = false;
                try {
                    const data = JSON.parse(r.responseText);
                    const rows = data?.data?.rows || [];
                    if (rows.length > 0) {
                        showFloatingBadge(`📦 Đang đồng bộ ${rows.length.toLocaleString('vi-VN')} đơn về Dashboard...`, '#0068a7');
                        sendToDashboard({ rows: rows, jwt_token: jwt, token: tk });
                    }
                } catch(e) {
                    console.error('Error parsing card 4885:', e);
                }
            },
            onerror: function(e) {
                window.__is_fetching_card_4885 = false;
                console.error('Error fetching card 4885:', e);
            }
        });
    }

    // Tự động bắt mọi gói tin qua Fetch
    const originalFetch = window.fetch;
    window.fetch = async function(...args) {
        const url = (args[0] && typeof args[0] === 'string') ? args[0] : (args[0]?.url || '');
        try {
            if (args[1] && args[1].headers) {
                let h = args[1].headers;
                let tk = h.token || (h.get && h.get('token')) || '';
                let rip = h['remote-ip'] || (h.get && h.get('remote-ip')) || '';
                if (tk) {
                    window.__GHN_TOKEN = tk;
                    window.__GHN_REMOTE_IP = rip;
                }
            }
        } catch(e) {}

        const res = await originalFetch.apply(this, args);

        try {
            const clone = res.clone();
            if (url.includes('metabase-token')) {
                clone.json().then(data => {
                    const jwt = data?.data?.token;
                    if (jwt) {
                        window.__GHN_JWT = jwt;
                        sendToDashboard({ token: window.__GHN_TOKEN, remote_ip: window.__GHN_REMOTE_IP, jwt_token: jwt });
                        fetchDirectCard4885(jwt, window.__GHN_TOKEN);
                    }
                }).catch(() => {});
            } else if (url.includes('card/4885') || url.includes('card/4884')) {
                clone.json().then(data => {
                    const rows = data?.data?.rows || [];
                    if (rows.length > 0) {
                        showFloatingBadge(`📦 Đang đồng bộ ${rows.length.toLocaleString('vi-VN')} đơn...`, '#0068a7');
                        sendToDashboard({ rows: rows, jwt_token: window.__GHN_JWT, token: window.__GHN_TOKEN });
                    }
                }).catch(() => {});
            }
        } catch(e) {}

        return res;
    };

    // Tự động bắt mọi gói tin qua XMLHttpRequest
    const originalXhrOpen = XMLHttpRequest.prototype.open;
    const originalXhrSend = XMLHttpRequest.prototype.send;
    const originalSetRequestHeader = XMLHttpRequest.prototype.setRequestHeader;

    XMLHttpRequest.prototype.setRequestHeader = function(header, value) {
        if (header.toLowerCase() === 'token') window.__GHN_TOKEN = value;
        if (header.toLowerCase() === 'remote-ip') window.__GHN_REMOTE_IP = value;
        return originalSetRequestHeader.apply(this, arguments);
    };

    XMLHttpRequest.prototype.open = function(method, url) {
        this._ghn_url = url;
        return originalXhrOpen.apply(this, arguments);
    };

    XMLHttpRequest.prototype.send = function() {
        this.addEventListener('load', function() {
            try {
                const url = this._ghn_url || '';
                if (url.includes('metabase-token')) {
                    const data = JSON.parse(this.responseText);
                    const jwt = data?.data?.token;
                    if (jwt) {
                        window.__GHN_JWT = jwt;
                        sendToDashboard({ token: window.__GHN_TOKEN, remote_ip: window.__GHN_REMOTE_IP, jwt_token: jwt });
                        fetchDirectCard4885(jwt, window.__GHN_TOKEN);
                    }
                } else if (url.includes('card/4885') || url.includes('card/4884')) {
                    const data = JSON.parse(this.responseText);
                    const rows = data?.data?.rows || [];
                    if (rows.length > 0) {
                        showFloatingBadge(`📦 Đang đồng bộ ${rows.length.toLocaleString('vi-VN')} đơn...`, '#0068a7');
                        sendToDashboard({ rows: rows, jwt_token: window.__GHN_JWT, token: window.__GHN_TOKEN });
                    }
                }
            } catch(e) {}
        });
        return originalXhrSend.apply(this, arguments);
    };

    // Quét chủ động
    async function activeScanner() {
        try {
            let tk = window.__GHN_TOKEN || localStorage.getItem('token') || localStorage.getItem('accessToken') || '';
            if (!tk) {
                for (let i = 0; i < localStorage.length; i++) {
                    const k = localStorage.key(i), v = localStorage.getItem(k);
                    if (v && v.length > 20) {
                        const m = v.match(/[a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12}/i);
                        if (m) { tk = m[0]; break; }
                    }
                }
            }
            if (tk) window.__GHN_TOKEN = tk;

            let jwt = window.__GHN_JWT;
            if (!jwt) {
                const iframes = document.querySelectorAll('iframe');
                for (const f of iframes) {
                    const m = (f.src || '').match(/dashboard\/([a-zA-Z0-9_\-\.]+)/);
                    if (m && m[1].length > 40) { jwt = m[1]; break; }
                }
            }
            if (!jwt && location.hostname === 'data-bi.ghn.vn') {
                const m = location.pathname.match(/dashboard\/([a-zA-Z0-9_\-\.]+)/);
                if (m && m[1].length > 40) jwt = m[1];
            }

            if (jwt) {
                window.__GHN_JWT = jwt;
                fetchDirectCard4885(jwt, tk);
            }
        } catch(e) {}
    }

    setTimeout(activeScanner, 1500);
    setTimeout(activeScanner, 4000);
    setInterval(activeScanner, 180000); // Mỗi 3 phút tự động quét và sync
})();
